class AgeError(Exception):
    pass


try:
    age = int(input("Enter your age: "))

    if age < 18:
        raise AgeError("Age must be 18 or above")

244:errorhandling quntityerror
class PriceError(Exception):
    pass
class QuantityError(Exception):
    pass
try:
    price = int(input("Enter product price: "))
    quantity = int(input("Enter product quantity: "))
    if not(price < 100 and price<1000):
        raise PriceError("Price must be 1 digit")
    elif quantity <= 0:
        raise QuantityError("Quantity must be greater than 0")
    else:
        print("Product price and quantity are valid.")
        print("Price:", price)
        print("Quantity:", quantity)
except PriceError as e:
    print("Price Error:", e)
except QuantityError as e:
    print("Quantity Error:", e)
finally:
    print("Success: Program execution completed.")
'''
Enter product price: 999
Enter product quantity: 2
Price Error: Price must be 1 digit
Success: Program execution completed.
'''
