class Employee:
    def __init__(self):
        self.emp_id = 0
        self.name = ""

    def get_data(self):
        self.emp_id = int(input("Enter Employee ID: "))
        self.name = input("Enter Employee Name: ")

    def put_data(self):
        print("Employee ID:", self.emp_id)
        print("Employee Name:", self.name)


class Attendance(Employee):
    def get_data(self):
        super().get_data()

        self.total = int(input("Enter Total Days: "))
        self.ptotal = int(input("Enter Total Present Days: "))

        self.atotal = self.total - self.ptotal

    def put_data(self):
        super().put_data()

        print("Total Days:", self.total)
        print("Present Days:", self.ptotal)
        print("Absent Days:", self.atotal)


class Salary(Attendance):
    def get_data(self):
        super().get_data()

        self.bs = int(input("Enter Basic Salary: "))

    def calculate_salary(self):

        if self.bs < 5000:
            self.hra = self.bs * 0.10
            self.da = self.bs * 0.20
            self.pf = self.bs * 0.05

        elif self.bs < 10000:
            self.hra = self.bs * 0.20
            self.da = self.bs * 0.40
            self.pf = self.bs * 0.10

        else:
            self.hra = self.bs * 0.30
            self.da = self.bs * 0.60
            self.pf = self.bs * 0.15

      
        self.ns = self.bs + self.hra + self.da - self.pf

        self.ns = (self.ns * self.ptotal) / self.total

    def put_data(self):
        super().put_data()

        print("Basic Salary:", self.bs)
        print("HRA:", self.hra)
        print("DA:", self.da)
        print("PF:", self.pf)
        print("Net Salary:", self.ns)

2. Multiple Inheritance

class Father:
    def show(self):
        print("Father class")

class Mother:
    def display(self):
        print("Mother class")

class Child(Father, Mother):
    def result(self):
        print("Child class")

c = Child()
c.show()
c.display()
c.result()

output::
Father class
Mother class
Child class

5. Hybrid Inheritance

class A:
    def show(self):
        print("Class A")

class B(A):
    def display(self):
        print("Class B")

class C(A):
    def result(self):
        print("Class C")

class D(B, C):
    def output(self):
        print("Class D")

d = D()
d.show()
d.display()
d.result()
d.output()

output::
Class A
Class B
Class C
Class D
